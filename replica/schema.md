# Schema: local stores (NotchCore)

No database and no server. Each store is a Codable array written atomically
(`Data.write(options: .atomic)`, file mode 0600) to
`~/Library/Application Support/MacbookNotch/`. All times are `Date` (UTC
instants), shown in the user's time zone. Money is integer minor units plus
an ISO 4217 code.

```swift
// MARK: Shelf (shelf.json + Shelf/<id>/<filename>)
struct ShelfItem: Codable, Identifiable, Hashable {
    let id: UUID
    var filename: String
    var storage: Storage            // copied file, or a bookmark for very large files
    var byteSize: Int64
    let addedAt: Date
    var expiresAt: Date             // addedAt + retention setting (default 24 h)
    enum Storage: Codable, Hashable {
        case copied(relativePath: String)
        case linked(bookmark: Data)
    }
}

// MARK: Clipboard (clipboard.json + Clips/<id>.png)
struct Clip: Codable, Identifiable, Hashable {
    let id: UUID
    var kind: Kind                  // text | url | image | fileURLs
    var text: String?               // text / url
    var imagePath: String?          // relative, for .image
    var fileURLs: [URL]?
    var sourceAppBundleID: String?
    let copiedAt: Date
    var contentHash: String         // SHA-256, for de-duplication
    enum Kind: String, Codable { case text, url, image, fileURLs }
}
// Never stored: items carrying org.nspasteboard.ConcealedType,
// org.nspasteboard.TransientType or org.nspasteboard.AutoGeneratedType, or
// copied while a deny-listed app is frontmost.

// MARK: Notes (notes.json)
struct Note: Codable, Identifiable { let id: UUID; var body: String; var updatedAt: Date }

// MARK: To-dos (todos.json)
struct TodoItem: Codable, Identifiable {
    let id: UUID; var title: String; var isDone: Bool
    let createdAt: Date; var completedAt: Date?
}

// MARK: Timers (timers.json)
struct TimerPreset: Codable, Identifiable { let id: UUID; var label: String; var seconds: Int }
struct ActiveTimer: Codable, Identifiable {
    let id: UUID; var label: String
    var endsAt: Date?               // set while running
    var remainingWhenPaused: TimeInterval?   // set while paused
}

// MARK: Stocks (stocks.json)
struct Ticker: Codable, Identifiable, Hashable {
    var id: String { symbol }
    var symbol: String
    var lastPriceMinor: Int?; var currency: String?
    var changePercent: Double?; var fetchedAt: Date?
}

// MARK: Weather (weather.json)
struct WeatherSnapshot: Codable {
    var locationName: String; var latitude: Double; var longitude: Double
    var temperatureC: Double; var conditionSymbol: String; var conditionText: String
    var highC: Double?; var lowC: Double?; var fetchedAt: Date
}

// MARK: AI usage (ai-usage.json)
struct ProviderUsage: Codable, Identifiable {
    var id: Provider { provider }
    var provider: Provider
    var mode: Mode                  // localLogs (default) | providerEndpoint (opt-in)
    var windows: [UsageWindow]      // e.g. 5-hour session, weekly
    var status: Status; var fetchedAt: Date?
    enum Provider: String, Codable, CaseIterable { case claude, codex, copilot, cursor, grok }
    enum Mode: String, Codable { case localLogs, providerEndpoint }
    enum Status: String, Codable { case ok, notInstalled, notSignedIn, unavailable }
}
struct UsageWindow: Codable { var label: String; var usedFraction: Double?; var tokens: Int?; var resetsAt: Date? }

// MARK: Revenue (revenue.json; API keys only in the Keychain)
struct RevenueSource: Codable, Identifiable {
    let id: UUID
    var kind: Kind
    var displayName: String         // e.g. the store name
    var keychainAccount: String     // key lives at service app.macbooknotch.revenue / this account
    var shopDomain: String?         // Shopify only
    enum Kind: String, Codable, CaseIterable { case stripe, shopify, lemonSqueezy, gumroad, dodo, polar, paddle }
}
struct RevenueSnapshot: Codable {
    var sourceID: UUID
    var day: String                 // yyyy-MM-dd in the user's time zone
    var totalsMinor: [String: Int]  // currency -> minor units; never summed across currencies
    var orderCount: Int; var fetchedAt: Date; var error: String?
}

// MARK: Teleprompter (scripts.json)
struct Script: Codable, Identifiable { let id: UUID; var title: String; var body: String; var wordsPerMinute: Int }

// MARK: Chat (memory only, never written to disk)
struct ChatMessage: Identifiable { let id: UUID; var role: Role; var text: String; enum Role { case user, assistant } }

// MARK: Licence (Keychain service app.macbooknotch.licence)
struct LicenceState: Codable {
    var key: String; var instanceID: String; var instanceName: String
    var activatedAt: Date; var lastValidatedAt: Date
    var status: Status
    enum Status: String, Codable { case active, invalid, disabled, expiredGrace }
}

// MARK: Settings (UserDefaults keys)
// tools.enabled [String], tools.order [String], hover.openDelayMs Int (100),
// hover.closeDelayMs Int (300), display.mode String (builtIn | all | mouse),
// shelf.retentionHours Int (24), clipboard.maxItems Int (200),
// clipboard.denyList [String], fullscreen.hide Bool (true), launchAtLogin Bool,
// volumeHUD.enabled Bool, aiUsage.mode.<provider> String
```
